Category table
-------------------- 
 CREATE TABLE "IN_CATEGORY" 
   (	"CT_ID" NUMBER, 
	"CT_NAME" VARCHAR2(50), 
	"PARENT_CODE" NUMBER, 
	"CT_CODE" VARCHAR2(100), 
	 PRIMARY KEY ("CT_ID")
  USING INDEX  ENABLE
   ) ;
   
 Item table
--------------------   
     CREATE TABLE "IN_ITEM" 
   (	"ITEM_NO" NUMBER(30,0), 
	"ITEM_NAME" VARCHAR2(100) NOT NULL ENABLE, 
	"ITEM_ID" VARCHAR2(50), 
	"CT_ID" NUMBER
   ) ;

Constraint
----------
  ALTER TABLE "IN_ITEM" ADD CONSTRAINT "FK_ITEM_CATEGORY" FOREIGN KEY ("CT_ID")
	  REFERENCES "IN_CATEGORY" ("CT_ID") ENABLE;
	  
PL/SQL Function Body returning a CLOB
--------------------   
DECLARE
    v_html CLOB;

    -- Recursive procedure to build the category and item tree
    PROCEDURE build_tree(p_parent_id IN NUMBER) IS

     BEGIN
     -- Fetch child categories for the current parent
        FOR cat_rec IN (
            SELECT 
                c.CT_ID, 
                c.CT_NAME,
                c.CT_CODE
            FROM 
                IN_CATEGORY c
            WHERE 
                NVL(c.PARENT_CODE, 0) = p_parent_id
            ORDER BY 
                c.CT_ID
        )

        LOOP
            -- Add category to the tree
            v_html := v_html || '<li class="category" data-category-id="' || cat_rec.CT_ID || '">';
            v_html := v_html || '<span class="toggle">+</span> ' || cat_rec.CT_CODE || '-' || cat_rec.CT_NAME;
            v_html := v_html || '<ul class="nested">';

            -- Fetch items directly under this category
            FOR item_rec IN (
                SELECT 
                    i.ITEM_NAME, 
                    i.ITEM_ID, 
                    i.ITEM_NO 
                FROM 
                    IN_ITEM i
                WHERE 
                    i.CT_ID = cat_rec.CT_ID -- Ensure proper mapping
                ORDER BY 
                   i.ITEM_NO,i.ITEM_ID asc
            )
            LOOP
                -- Add items to the tree
                v_html := v_html || '<li class="item">' || item_rec.ITEM_ID || ' - ' || item_rec.ITEM_NAME || '</li>';
            END LOOP;


             -- Recursive call for child categories
            build_tree(cat_rec.CT_ID);

            -- Close the category list
            v_html := v_html || '</ul>';
            v_html := v_html || '</li>';
        END LOOP;
    END;


    BEGIN
    -- Initialize the HTML
    v_html := '<ul id="categoryTree">';

    -- Start building the tree from the root categories (where PARENT_CODE IS NULL or 0)
    build_tree(0);

    -- Close the HTML structure
    v_html := v_html || '</ul>';

    -- Return the HTML content
    RETURN v_html;
END;

Function and Global Variable Declaration

document.addEventListener('DOMContentLoaded', function () {
    // Toggle functionality for expand/collapse
    document.querySelectorAll('#categoryTree .toggle').forEach(function (toggle) {
        toggle.addEventListener('click', function () {
            const nested = this.parentElement.querySelector('.nested');
            if (nested) {
                nested.classList.toggle('active');
                this.textContent = nested.classList.contains('active') ? '−' : '+';
            }
        });
    });

    // Expand All Button
    document.getElementById('expandAll').addEventListener('click', function (event) {
        event.preventDefault(); // Prevent the default behavior
        document.querySelectorAll('#categoryTree .nested').forEach(function (nested) {
            nested.classList.add('active'); // Expand all nested lists
        });
        document.querySelectorAll('#categoryTree .toggle').forEach(function (toggle) {
            toggle.textContent = '−'; // Set all toggle buttons to "−"
        });
    });

    // Collapse All Button
    document.getElementById('collapseAll').addEventListener('click', function (event) {
        event.preventDefault(); // Prevent the default behavior
        document.querySelectorAll('#categoryTree .nested').forEach(function (nested) {
            nested.classList.remove('active'); // Collapse all nested lists
        });
        document.querySelectorAll('#categoryTree .toggle').forEach(function (toggle) {
            toggle.textContent = '+'; // Set all toggle buttons to "+"
        });
    });

// Search functionality
    const searchInput = document.getElementById('treeSearch');
    searchInput.addEventListener('input', function () {
        const query = searchInput.value.toLowerCase();
        const treeItems = document.querySelectorAll('#categoryTree li');

        treeItems.forEach(function (item) {
            const text = item.textContent.toLowerCase();
            if (text.includes(query)) {
                item.style.display = 'flex'; // Show matching items
                let parent = item.parentElement;
                while (parent && parent.id !== 'categoryTree') {
                    if (parent.classList.contains('nested')) {
                        parent.classList.add('active'); // Expand parent category
                    }
                    parent = parent.parentElement;
                }
            } else {
                item.style.display = 'none'; // Hide non-matching items
            }
        });
    });
});
